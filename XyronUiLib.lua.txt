local THEME = {
	Background   = Color3.fromRGB(15, 13, 22),
	TitleBar     = Color3.fromRGB(22, 19, 32),
	Accent       = Color3.fromRGB(140, 60, 255),
	AccentHover  = Color3.fromRGB(165, 90, 255),
	DarkContainer= Color3.fromRGB(25, 22, 36),
	TextBright   = Color3.fromRGB(255, 255, 255),
	TextMuted    = Color3.fromRGB(160, 155, 180),
	Stroke       = Color3.fromRGB(55, 40, 85)
}

local function tween(object, info, properties)
	local tw = TweenService:Create(object, info, properties)
	tw:Play()
	return tw
end

local function applyCorner(parent, radius)
	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, radius)
	corner.Parent = parent
	return corner
end

local function applyStroke(parent, color, thickness)
	local stroke = Instance.new("UIStroke")
	stroke.Color = color or THEME.Stroke
	stroke.Thickness = thickness or 1
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = parent
	return stroke
end

local function makeDraggable(guiObject)
	local dragging, dragInput, dragStart, startPos
	guiObject.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = guiObject.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	guiObject.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			guiObject.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

	function Window:CreateTab(tabName)
		local isFirst = (self.CurrentTab == nil)
		
		local tabBtn = Instance.new("TextButton")
		tabBtn.Size = UDim2.new(0.33, -5, 1, 0)
		tabBtn.BackgroundTransparency = 1
		tabBtn.Text = tabName
		tabBtn.TextColor3 = isFirst and THEME.TextBright or THEME.TextMuted
		tabBtn.TextSize = 12
		tabBtn.Font = Enum.Font.GothamMedium
		tabBtn.Parent = tabBarContainer

		local underline = Instance.new("Frame")
		underline.Size = isFirst and UDim2.new(0.8, 0, 0, 2) or UDim2.new(0, 0, 0, 2)
		underline.Position = UDim2.new(0.1, 0, 1, -2)
		underline.BackgroundColor3 = THEME.Accent
		underline.BorderSizePixel = 0
		underline.Parent = tabBtn

		local page = Instance.new("ScrollingFrame")
		page.Size = UDim2.new(1, 0, 1, 0)
		page.Position = isFirst and UDim2.new(0, 0, 0, 0) or UDim2.new(1, 0, 0, 0)
		page.BackgroundTransparency = 1
		page.ScrollBarThickness = 3
		page.ScrollBarImageColor3 = THEME.Accent
		page.AutomaticCanvasSize = Enum.AutomaticSize.Y
		page.Visible = isFirst
		page.Parent = pagesFolder

		local pagePadding = Instance.new("UIPadding")
		pagePadding.PaddingLeft = UDim.new(0, 2)
		pagePadding.PaddingRight = UDim.new(0, 8)
		pagePadding.PaddingTop = UDim.new(0, 2)
		pagePadding.PaddingBottom = UDim.new(0, 6)
		pagePadding.Parent = page

		local pageLayout = Instance.new("UIListLayout")
		pageLayout.SortOrder = Enum.SortOrder.LayoutOrder
		pageLayout.Padding = UDim.new(0, 8)
		pageLayout.Parent = page

		local tabData = {Button = tabBtn, Page = page, Underline = underline, Name = tabName}
		table.insert(self.Tabs, tabData)

		if isFirst then self.CurrentTab = tabData end

		tabBtn.MouseButton1Click:Connect(function()
			if self.CurrentTab == tabData or self.IsSwitchingTab then return end
			self.IsSwitchingTab = true

			local oldTab = self.CurrentTab
			local newTab = tabData
			
			local oldIndex, newIndex
			for i, t in ipairs(self.Tabs) do
				if t == oldTab then oldIndex = i end
				if t == newTab then newIndex = i end
			end

			local slideDirection = newIndex > oldIndex and 1 or -1
			self.CurrentTab = tabData

			tween(oldTab.Button, TweenInfo.new(0.2), {TextColor3 = THEME.TextMuted})
			tween(oldTab.Underline, TweenInfo.new(0.2), {Size = UDim2.new(0, 0, 0, 2), Position = UDim2.new(0.5, 0, 1, -2)})
			tween(newTab.Button, TweenInfo.new(0.2), {TextColor3 = THEME.TextBright})
			tween(newTab.Underline, TweenInfo.new(0.2), {Size = UDim2.new(0.8, 0, 0, 2), Position = UDim2.new(0.1, 0, 1, -2)})

			newTab.Page.Position = UDim2.new(slideDirection, 0, 0, 0)
			newTab.Page.Visible = true

			tween(oldTab.Page, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Position = UDim2.new(-slideDirection, 0, 0, 0)})
			local pageTween = tween(newTab.Page, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Position = UDim2.new(0, 0, 0, 0)})

			pageTween.Completed:Connect(function()
				oldTab.Page.Visible = false
				self.IsSwitchingTab = false
			end)
		end)

			frame.BackgroundColor3 = THEME.DarkContainer
			frame.Parent = page

			applyCorner(knob, 6)

			local function updateSlider(input)
				local posX = input.Position.X - track.AbsolutePosition.X
				local percentage = math.clamp(posX / track.AbsoluteSize.X, 0, 1)
				local value = math.floor(min + (max - min) * percentage)
				valLabel.Text = tostring(value)
				tween(fill, TweenInfo.new(0.05), {Size = UDim2.new(percentage, 0, 1, 0)})

					dragging = true; updateSlider(input)
				end
			end)

			UserInputService.InputChanged:Connect(function(input)
				if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
					updateSlider(input)
				end
			end)

			UserInputService.InputEnded:Connect(function(input)
				if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
					dragging = false
				end
			end)
		end

				isOpen = not isOpen
				local targetSize = isOpen and UDim2.new(1, 0, 0, 36 + (#options * 30)) or UDim2.new(1, 0, 0, 36)

							selectedMulti[opt] = nil
							tween(leftIndicator, TweenInfo.new(0.2), {Size = UDim2.new(0, 3, 0, 0), Position = UDim2.new(0, 0, 0.5, 0)})
							tween(optBtn, TweenInfo.new(0.2), {TextColor3 = THEME.TextMuted})
						else
							selectedMulti[opt] = true
							tween(leftIndicator, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = UDim2.new(0, 3, 0, 16), Position = UDim2.new(0, 0, 0.5, -8)})
							tween(optBtn, TweenInfo.new(0.2), {TextColor3 = THEME.TextBright})
						end
						local result = {}
						for k, _ in pairs(selectedMulti) do table.insert(result, k) end
						if callback then callback(result) end
					else
						for _, data in ipairs(optionData) do
							tween(data.btn, TweenInfo.new(0.2), {TextColor3 = THEME.TextMuted})
							tween(data.indicator, TweenInfo.new(0.2), {Size = UDim2.new(0, 3, 0, 0), Position = UDim2.new(0, 0, 0.5, 0)})
						end
						tween(leftIndicator, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = UDim2.new(0, 3, 0, 16), Position = UDim2.new(0, 0, 0.5, -8)})
						tween(optBtn, TweenInfo.new(0.2), {TextColor3 = THEME.TextBright})
						headerBtn.Text = "   " .. text .. ": " .. opt
						isOpen = false
						tween(frame, TweenInfo.new(0.3, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Size = UDim2.new(1, 0, 0, 36)})
						tween(icon, TweenInfo.new(0.3), {Rotation = 0})
						if callback then callback(opt) end
					end
				end)
			end
		end

		return TabFunctions
	end

	return Window
end
